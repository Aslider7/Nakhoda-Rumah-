    function loadState() {
      try {
        const raw = localStorage.getItem(APP_CONFIG.storageKey);
        if (!raw) return defaultState();
        const parsed = JSON.parse(raw);
        const merged = { ...defaultState(), ...parsed };
        const today = getTodayKey();

        // Kira Hari Bangkit (tanpa menghukum jika terlepas)
        if (merged.lastActiveDate !== today) {
          merged.daysActive = (merged.daysActive || 1) + 1;
          merged.lastActiveDate = today;
        }
        // Reset kaunter harian jika hari baru
        if (merged.tasbihTodayDate !== today) {
          merged.tasbihTodayDate = today;
          merged.tasbihTodayCount = 0;
        }
        if (merged.todayTasksDate !== today) {
          merged.todayTasksDate = today;
          merged.todayTasks = [];
        }
        if (merged.prayerBlocksDate !== today) {
          merged.prayerBlocksDate = today;
          Object.keys(merged.prayerBlocks).forEach(k => {
            merged.prayerBlocks[k].done = false;
          });
        }
        return merged;
      } catch (e) {
        return defaultState();
      }
    }

    let audioCtx = null;
    function playSoftClick(freq = 520, duration = 0.045) {
      if (!state.tasbihSound) return;
      try {
        if (!audioCtx) audioCtx = new (window.AudioContext || window.webkitAudioContext)();
        if (audioCtx.state === "suspended") audioCtx.resume();
        const osc = audioCtx.createOscillator();
        const gain = audioCtx.createGain();
        osc.type = "sine";
        osc.frequency.setValueAtTime(freq, audioCtx.currentTime);
        gain.gain.setValueAtTime(0.06, audioCtx.currentTime);
        gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + duration);
        osc.connect(gain);
        gain.connect(audioCtx.destination);
        osc.start();
        osc.stop(audioCtx.currentTime + duration);
      } catch (e) {}
    }

    function renderTodayTasks() {
      const list = document.getElementById("todayTaskList");
      const countBadge = document.getElementById("taskCountBadge");
      const inputRow = document.getElementById("taskInputRow");
      const limitMsg = document.getElementById("taskLimitMsg");

      const doneCount = state.todayTasks.filter(t => t.done).length;
      countBadge.textContent = `${doneCount} / ${Math.max(3, state.todayTasks.length)}`;

      if (state.todayTasks.length === 0) {
        list.innerHTML = `<div class="card-soft muted" style="text-align:center;">Belum ada tugasan hari ini. Tulis 1 perkara paling mudah dulu untuk mulakan momentum.</div>`;
      } else {
        list.innerHTML = state.todayTasks.map((t, i) => `
          <div class="task-item ${t.done ? "done" : ""}">
            <div class="check-circle" onclick="toggleTodayTask(${i})">${t.done ? "✓" : ""}</div>
            <div class="task-text">${escapeHtml(t.text)}</div>
            <div class="task-actions">
              ${!t.done ? `<button class="icon-btn" title="Mula 5 minit" onclick="focusOnTask(${i})">⏱</button>` : ""}
              <button class="icon-btn" title="Padam" onclick="deleteTodayTask(${i})">✕</button>
            </div>
          </div>
        `).join("");
      }

      const atLimit = state.todayTasks.length >= 3;
      inputRow.style.display = atLimit ? "none" : "flex";
      limitMsg.style.display = atLimit ? "block" : "none";
      updateHeaderStats();
    }

    function addTodayTask() {
      const inp = document.getElementById("newTaskInput");
      const val = inp.value.trim();
      if (!val) return showToast("Tulis tugasan ringkas dahulu.");
      if (state.todayTasks.length >= 3) return showToast("Cukup 3 fokus utama hari ini.");
      state.todayTasks.push({ text: val, done: false });
      inp.value = "";
      saveState();
      renderTodayTasks();
      showToast("Bagus! Satu fokus sudah jelas.");
    }

    function toggleTodayTask(i) {
      state.todayTasks[i].done = !state.todayTasks[i].done;
      if (state.todayTasks[i].done) {
        playSoftClick(660, 0.08);
        showToast("Alhamdulillah, satu langkah kecil selesai!");
      }
      saveState();
      renderTodayTasks();
    }

    function deleteTodayTask(i) {
      state.todayTasks.splice(i, 1);
      saveState();
      renderTodayTasks();
    }

    function focusOnTask(i) {
      const task = state.todayTasks[i];
      document.getElementById("activeTimerTaskLabel").textContent = "Fokus sekarang: " + task.text;
      selectTimerMinutes(5);
      if (!timerRunning) toggleTimer();
      showToast("Pemasa 5 minit dimulakan. Buat perlahan-lahan.");
    }

    let timerSelectedMin = 5;
    let timerSecondsLeft = 300;
    let timerRunning = false;
    let timerInterval = null;

    function selectTimerMinutes(min) {
      if (min > 5 && !state.isPro) {
        openModal("proModal");
        return showToast("Pemasa 15 & 25 minit tersedia dalam versi Pro.");
      }
      stopTimer();
      timerSelectedMin = min;
      timerSecondsLeft = min * 60;
      document.querySelectorAll("[id^='preset-']").forEach(el => el.classList.remove("active"));
      document.getElementById("preset-" + min).classList.add("active");
      updateTimerDisplay();
      document.getElementById("timerStatus").textContent = `Sedia untuk fokus ${min} minit dengan tenang.`;
    }

    function updateTimerDisplay() {
      const m = String(Math.floor(timerSecondsLeft / 60)).padStart(2, "0");
      const s = String(timerSecondsLeft % 60).padStart(2, "0");
      document.getElementById("timerClock").textContent = `${m}:${s}`;
    }

    function toggleTimer() {
      if (timerRunning) {
        stopTimer();
        document.getElementById("timerStatus").textContent = "Dijeda sebentar. Sambung bila sedia.";

    function exportDataJSON() {
      if (!state.isPro) {
        closeModal("settingsModal");
        openModal("proModal");
        return showToast("Fungsi Backup JSON tersedia untuk ahli Pro.");
      }
      const blob = new Blob([JSON.stringify(state, null, 2)], { type: "application/json" });
      const url = URL.createObjectURL(blob);
      const a = document.createElement("a");
      a.href = url;
      a.download = `LangkahBaru_Backup_${getTodayKey()}.json`;
      a.click();
      URL.revokeObjectURL(url);
      showToast("Fail backup berjaya dimuat turun.");
    }

    function importDataJSON(e) {
      const file = e.target.files[0];
      if (!file) return;
      const reader = new FileReader();
      reader.onload = () => {
        try {
          const parsed = JSON.parse(reader.result);
          state = { ...defaultState(), ...parsed };
          saveState();
          renderAll();
          closeModal("settingsModal");
          showToast("Data backup berjaya dipulihkan!");
        } catch (err) {
          showToast("Fail backup tidak sah.");
        }
      };
      reader.readAsText(file);
    }

      state = defaultState();

      return String(str || "")
        .replace(/&/g, "&amp;")
        .replace(/</g, "&lt;")
        .replace(/>/g, "&gt;")
        .replace(/"/g, "&quot;");
    }
