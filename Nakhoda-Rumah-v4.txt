let audioCtx=null;
function initAudio(){if(!audioCtx)audioCtx=new(window.AudioContext||window.webkitAudioContext)();if(audioCtx.state==='suspended')audioCtx.resume();}
function playChime(){
  try{initAudio();[523.25,659.25,783.99].forEach((f,i)=>{
    const o=audioCtx.createOscillator(),g=audioCtx.createGain();
    o.type='sine';o.frequency.value=f;
    g.gain.setValueAtTime(0.001,audioCtx.currentTime+i*0.22);
    g.gain.exponentialRampToValueAtTime(0.18,audioCtx.currentTime+i*0.22+0.04);
    g.gain.exponentialRampToValueAtTime(0.001,audioCtx.currentTime+i*0.22+0.7);
    o.connect(g);g.connect(audioCtx.destination);
    o.start(audioCtx.currentTime+i*0.22);o.stop(audioCtx.currentTime+i*0.22+0.75);
  });}catch(e){}
}

}
